      \begin{enumerate}\setlength\itemsep{2pt}\footnotesize
        \item ChaosMiddleware (drop / delay / tamper)
        \item Auth layer (API-key validation)
        \item HMAC-SHA256 verify (constant-time)
        \item AES-256-GCM decrypt (if encrypted)
        \item Pydantic validation
        \item SQLite persist
        \item WebSocket broadcast
        \item Ollama AI task (fire-and-forget)
      \end{enumerate}

% SLIDE 7 — Methodology: Backend & Communications

\begin{frame}{Methodology: Backend \& Communications}
  \begin{columns}[c, onlytextwidth]
    \column{0.55\textwidth}
      \begin{cvisbox}[title={Security Mechanisms}]
        \begin{itemize}\setlength\itemsep{4pt}\footnotesize
          \item \textbf{Auth:} API key per device (SHA-256 hashed).
          \item \textbf{Integrity:} HMAC-SHA256, constant-time verify.
          \item \textbf{Encryption:} AES-256-GCM, fresh random IV.
          \item \textbf{Tamper:} Chaos mutates body $\rightarrow$ HMAC fail $\rightarrow$ 401.
          \item \textbf{Replay:} (device\_id, timestamp) dedup.
          \item \textbf{Revocation:} Disconnect sets \texttt{active=0}.
          \item \textbf{Audit:} All events logged in \texttt{auth\_logs} table.
        \end{itemize}
      \end{cvisbox}

    \column{0.40\textwidth}
      \centering
      \imgplaceholder{4.8cm}{5.0cm}{Crypto Flow Diagram:\\Payload -> HMAC-SHA256\\-> AES-256-GCM\\-> Verify on server}

        \item \textbf{Auto-recommendation} per telemetry update.
        \item \textbf{Driver chat} grounded in current telemetry.
        \item AI kill-switch --- secure comms stays independent.
